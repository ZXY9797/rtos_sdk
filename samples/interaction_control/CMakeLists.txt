cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(interaction_control LANGUAGES CXX)
    enable_testing()
endif()
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${SDK_ROOT}/tools/cmake/modules/sdk_component.cmake")
foreach(component algo motor)
    if(NOT TARGET ${component})
        add_subdirectory("${SDK_ROOT}/component/${component}" "${CMAKE_CURRENT_BINARY_DIR}/${component}")
    endif()
endforeach()
add_executable(interaction_control main.cc)
target_link_libraries(interaction_control PRIVATE sdk::motor)
target_compile_features(interaction_control PRIVATE cxx_std_17)
if(COMMAND sdk_host_test)
    sdk_host_test(interaction_control interaction_control "motor")
else()
    add_test(NAME interaction_control COMMAND interaction_control)
endif()
